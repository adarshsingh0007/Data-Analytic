CREATE DATABASE zipbasket3;

USE zipbasket3;

CREATE TABLE orders (
  order_id      INT PRIMARY KEY,
  customer_id   INT  NOT NULL,
  product_id    INT  NOT NULL,
  order_date    DATE NOT NULL,
  quantity      INT  NOT NULL CHECK (quantity > 0),
  order_status  VARCHAR(15) NOT NULL CHECK (order_status IN ('Delivered','Cancelled')),
  delivery_mins INT
);
